cmake_minimum_required(VERSION 3.24)
project(meditor VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
    add_compile_options(/utf-8)
endif()

find_package(nlohmann_json CONFIG REQUIRED)
add_library(meditor_project src/project.cpp)
target_include_directories(meditor_project PUBLIC src)
target_link_libraries(meditor_project PUBLIC nlohmann_json::nlohmann_json)

option(MEDITOR_BUILD_APP "Build the editor" ON)
if(MEDITOR_BUILD_APP)
    find_package(Threads REQUIRED)
    find_package(SDL2 CONFIG REQUIRED)
    find_package(wxWidgets CONFIG QUIET)
    if(TARGET wx::core)
        set(wx_targets wx::core wx::base)
    else()
        find_package(wxWidgets REQUIRED COMPONENTS core base)
        include(${wxWidgets_USE_FILE})
        set(wx_targets ${wxWidgets_LIBRARIES})
    endif()
    if(VCPKG_TARGET_TRIPLET)
        find_package(FFMPEG REQUIRED)
        add_library(av_dependencies INTERFACE)
        target_include_directories(av_dependencies INTERFACE ${FFMPEG_INCLUDE_DIRS})
        target_link_directories(av_dependencies INTERFACE ${FFMPEG_LIBRARY_DIRS})
        target_link_libraries(av_dependencies INTERFACE ${FFMPEG_LIBRARIES})
    else()
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(av_dependencies REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale)
        set(av_target PkgConfig::av_dependencies)
    endif()
    if(NOT av_target)
        set(av_target av_dependencies)
    endif()
    add_library(meditor_media src/video.cpp src/audio.cpp)
    target_link_libraries(meditor_media PUBLIC meditor_project PRIVATE ${av_target} SDL2::SDL2 Threads::Threads)
    add_executable(meditor WIN32 src/app.cpp src/editor_view.cpp)
    target_link_libraries(meditor PRIVATE meditor_media ${wx_targets})
    if(WIN32)
        target_sources(meditor PRIVATE src/windows_screenshot.cpp)
        target_link_libraries(meditor PRIVATE dwmapi gdi32 user32)
    endif()
    install(TARGETS meditor RUNTIME DESTINATION .)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(project_test tests/project_test.cpp)
    target_link_libraries(project_test PRIVATE meditor_project)
    add_test(NAME project COMMAND project_test)
endif()

add_executable(meditor_fixture tools/fixture.cpp)
target_link_libraries(meditor_fixture PRIVATE meditor_project)

option(MEDITOR_TEST_MEDIA "Generate an FFmpeg fixture and test decode and audio" OFF)
if(BUILD_TESTING AND MEDITOR_BUILD_APP AND MEDITOR_TEST_MEDIA)
    add_executable(media_test tests/media_test.cpp)
    target_link_libraries(media_test PRIVATE meditor_media)
    set(media_fixture "${CMAKE_CURRENT_BINARY_DIR}/media-fixture")
    add_test(NAME media_fixture COMMAND ${CMAKE_COMMAND}
        -DOUTPUT=${media_fixture} -DFIXTURE=$<TARGET_FILE:meditor_fixture>
        -P ${CMAKE_CURRENT_SOURCE_DIR}/tools/fixture.cmake)
    set_tests_properties(media_fixture PROPERTIES FIXTURES_SETUP media_fixture TIMEOUT 120)
    add_test(NAME media COMMAND media_test "${media_fixture}/fixture.json")
    set_tests_properties(media PROPERTIES FIXTURES_REQUIRED media_fixture TIMEOUT 20 ENVIRONMENT "SDL_AUDIODRIVER=dummy")
endif()
